import { Component, OnInit } from '@angular/core';
import { ActivatedRoute } from '@angular/router';
import { QuizTile, SELBSTSTUDIUM_TILES, QUIZ_TILES, SPIELE_TILES } from '../../../shared/katalog';

interface Kategorie {
  id: string;
  eyebrow: string;
  title: string;
  note: string;
  tiles: QuizTile[];
}

const KATEGORIEN: Kategorie[] = [
  {
    id: 'selbststudium',
    eyebrow: 'Zum Nachlesen',
    title: 'Selbststudium',
    note: 'Lesestoff zum Aneignen und Nachschlagen — kein Frage-Antwort-Format.',
    tiles: SELBSTSTUDIUM_TILES,
  },
  {
    id: 'quiz',
    eyebrow: 'Wissen abfragen',
    title: 'Quiz',
    note: 'Wissen festigen und abklopfen — Fragen beantworten, sofort Rueckmeldung bekommen.',
    tiles: QUIZ_TILES,
  },
  {
    id: 'spiele',
    eyebrow: 'Ausprobieren',
    title: 'Spiele',
    note: 'Lernspiele statt Frage-Antwort — nutzen dieselben Themen, andere Form.',
    tiles: SPIELE_TILES,
  },
];

@Component({
  selector: 'app-kategorie',
  templateUrl: './kategorie.page.html',
  styleUrls: ['./kategorie.page.scss'],
  standalone: false,
})
export class KategoriePage implements OnInit {
  kategorie: Kategorie = KATEGORIEN[0];

  constructor(private readonly route: ActivatedRoute) {}

  ngOnInit(): void {
    this.route.paramMap.subscribe((params) => {
      const id = params.get('id');
      this.kategorie = KATEGORIEN.find((k) => k.id === id) ?? KATEGORIEN[0];
    });
  }

  trackByTileId(_index: number, tile: QuizTile): string {
    return tile.id;
  }

  tileImageStyle(tile: QuizTile): string | null {
    return tile.image ? `url('${tile.image}')` : null;
  }
}

=== AUSSERDEM: src/app/app-routing.module.ts anpassen ===
Fuege im Array `routes` einen neuen Eintrag ein, direkt VOR dem Eintrag mit `path: 'bildnachweise'`:

  {
    path: 'kategorie/:id',
    loadChildren: () => import('./modules/kategorie/kategorie.module')
      .then(m => m.KategorieModule)
  },

Aendere sonst NICHTS an app-routing.module.ts.

Danach: `npx ng build --configuration production` im Projekt-Root ausfuehren und pruefen, dass es fehlerfrei durchlaeuft.
