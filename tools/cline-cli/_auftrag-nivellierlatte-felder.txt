AENDERUNG 3 — buildRodCells() komplett ersetzen (Datei .ts):
Ersetze die komplette Methode `buildRodCells()` durch:

  private buildRodCells(): RodCell[] {
    const fieldH = 2;
    const fieldsPerDecimeter = 5;
    const decimeterCount = (this.rodMaxM - this.rodMinM) * 10;
    const toothWidth = 5.5;
    const notchWidth = 8;
    const notchHeight = fieldH / 3;
    const cells: RodCell[] = [];

    for (let d = 0; d < decimeterCount; d++) {
      const activeLeft = d % 2 === 0;
      const activeX = activeLeft ? 10 : 22;
      const toothX = activeLeft ? 22 : 10;
      const outerRedX = activeLeft ? 10 : 30;
      const decBaseY = d * 10;

      for (let f = 0; f < fieldsPerDecimeter; f++) {
        const fieldBaseY = decBaseY + f * fieldH;

        if (f === 0) {
          cells.push({ svgY: fieldBaseY, x: toothX, width: toothWidth, height: fieldH * 0.15 });
          const tooth2Y = fieldBaseY + fieldH * 0.5;
          cells.push({ svgY: tooth2Y, x: toothX, width: toothWidth, height: fieldH * 0.33 });
        } else {
          const notchAtBottom = (f - 1) % 2 === 0;
          if (notchAtBottom) {
            cells.push({ svgY: fieldBaseY, x: outerRedX, width: 4, height: notchHeight });
            cells.push({ svgY: fieldBaseY + notchHeight, x: activeX, width: 12, height: fieldH - notchHeight });
          } else {
            cells.push({ svgY: fieldBaseY, x: activeX, width: 12, height: fieldH - notchHeight });
            cells.push({ svgY: fieldBaseY + fieldH - notchHeight, x: outerRedX, width: 4, height: notchHeight });
          }
        }
      }
    }
    return cells;
  }

AENDERUNG 4 — buildDecimeterLabels() komplett ersetzen (Datei .ts):
Ersetze die komplette Methode `buildDecimeterLabels()` durch:

  private buildDecimeterLabels(): DecimeterLabel[] {
    const decimeterCount = (this.rodMaxM - this.rodMinM) * 10;
    const labels: DecimeterLabel[] = [];
    for (let d = 0; d < decimeterCount; d++) {
      const activeLeft = d % 2 === 0;
      const decBaseY = d * 10;
      const heightM = this.rodMaxM - (decBaseY + 10) / 100;
      const value = Math.round(heightM * 10);
      labels.push({ svgY: decBaseY + 1.4, x: activeLeft ? 16 : 28, text: String(value).padStart(2, '0') });
    }
    return labels;
  }

AENDERUNG 5 — Neue Methode buildDecimeterTicks() hinzufuegen (Datei .ts):
Fuege DIREKT NACH der Methode `buildDecimeterLabels()` (also vor der abschliessenden `}` der Klasse) eine neue Methode ein:

  private buildDecimeterTicks(): number[] {
    const decimeterCount = (this.rodMaxM - this.rodMinM) * 10;
    const ticks: number[] = [];
    for (let d = 0; d <= decimeterCount; d++) {
      ticks.push(d * 10);
    }
    return ticks;
  }
