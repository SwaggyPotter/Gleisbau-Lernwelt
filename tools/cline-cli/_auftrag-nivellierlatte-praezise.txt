AENDERUNG 1 — Praezise Latten-Geometrie (Datei .ts):
Ersetze die komplette Methode `buildRodCells()` durch:

  private buildRodCells(): RodCell[] {
    const decimeterCount = (this.rodMaxM - this.rodMinM) * 10;
    const cells: RodCell[] = [];
    for (let d = 0; d < decimeterCount; d++) {
      const spineX = d % 2 === 1 ? 22 : 10;
      const armX = d % 2 === 1 ? 10 : 22;
      const baseY = d * 10;
      cells.push({ svgY: baseY + 1, x: armX + 3.5, width: 5 });
      cells.push({ svgY: baseY + 3, x: armX + 3.5, width: 5 });
      cells.push({ svgY: baseY + 5, x: 10, width: 24 });
      cells.push({ svgY: baseY + 6, x: spineX, width: 12 });
      cells.push({ svgY: baseY + 7, x: 10, width: 24 });
      cells.push({ svgY: baseY + 8, x: spineX, width: 12 });
      cells.push({ svgY: baseY + 9, x: 10, width: 24 });
    }
    return cells;
  }

(Das Interface `RodCell` mit den Feldern `svgY`, `x`, `width` bleibt unveraendert stehen — nur der INHALT der Methode aendert sich. Jede Zelle ist jetzt 1cm hoch statt 2cm.)
